# Daily Cleanup Tick
country_event = {
	id = 99000
	title = "Daily Cleanup Tick"
	desc = "EVTDESC99000"
	picture = "production"
	
	is_triggered_only = yes
	
	option = {
		name = "Daily Cleanup Tick"

        # Break Xenon hivemind alliances
        any_country = {
            limit = {
                has_country_modifier = xenon_hivemind
            }
            country_event = 99001
        }

        # Break Khaak hivemind alliances
        any_country = {
            limit = {
                has_country_modifier = khaak_hivemind
            }
            country_event = 99002
        }

        # Retrigger tomorrow
        owner = { country_event = { id = 99000 days = 1 } }
    }
}
